/* ---------- resorte (damping + response, como los de Apple) ----------
   Siempre parte del valor y la velocidad actuales: se puede
   interrumpir y redirigir en cualquier momento sin saltos. */
class Spring {
  constructor(value, onUpdate){ this.x = value; this.v = 0; this.target = value; this.raf = 0; this.onUpdate = onUpdate; }
  to(target, { response = .4, damping = 1, velocity, onRest, always = false } = {}){
    this.target = target; this.onRest = onRest;
    if(velocity !== undefined) this.v = velocity;
    if(reduceMotion.matches && !always){ this.set(target); onRest && onRest(); return; }
    this.k = Math.pow(2 * Math.PI / response, 2);
    this.c = 4 * Math.PI * damping / response;
    if(!this.raf){ this.last = performance.now(); this.raf = requestAnimationFrame(this.step); }
  }
  step = now => {
    const dt = Math.min(.064, (now - this.last) / 1000); this.last = now;
    const n = Math.max(1, Math.ceil(dt / .004)), h = dt / n;
    for(let i = 0; i < n; i++){
      const a = -this.k * (this.x - this.target) - this.c * this.v;
      this.v += a * h; this.x += this.v * h;
    }
    if(Math.abs(this.v) < .5 && Math.abs(this.x - this.target) < .3){
      this.x = this.target; this.v = 0; this.raf = 0; this.onUpdate(this.x);
      const done = this.onRest; this.onRest = null; done && done(); return;
    }
    this.onUpdate(this.x);
    this.raf = requestAnimationFrame(this.step);
  };
  stop(){ cancelAnimationFrame(this.raf); this.raf = 0; }
  set(x){ this.stop(); this.x = x; this.v = 0; this.onUpdate(x); }
}
// dónde terminaría un gesto lanzado con esa velocidad (fórmula de Apple)
const project = (v, d = .998) => (v / 1000) * d / (1 - d);
// resistencia progresiva en los bordes
const rubberband = (over, dim, c = .55) => (over * dim * c) / (dim + c * Math.abs(over));
// velocidad a partir de los últimos movimientos del dedo
function tracker(){
  let pts = [];
  return {
    reset(){ pts = []; },
    add(x, t){ pts.push([x, t]); while(pts.length > 2 && t - pts[0][1] > 100) pts.shift(); },
    v(){ if(pts.length < 2) return 0; const [x0, t0] = pts[0], [x1, t1] = pts[pts.length - 1]; return t1 > t0 ? (x1 - x0) / (t1 - t0) * 1000 : 0; }
  };
}
/* scroll suave con resorte: baja fluido, se puede cortar con el dedo o la rueda en cualquier momento */
(function(){
  let spring = null;
  const stop = () => spring && spring.stop();
  ['wheel', 'touchstart', 'keydown', 'pointerdown'].forEach(t => addEventListener(t, stop, { passive:true }));
  window.smoothScrollTo = y => {
    y = Math.max(0, Math.min(y, document.documentElement.scrollHeight - innerHeight));
    if(!spring) spring = new Spring(scrollY, v => scrollTo(0, v));
    spring.stop(); spring.x = scrollY; spring.v = 0;
    const dist = Math.abs(y - scrollY);
    // siempre desliza, aunque el sistema tenga "reducir movimiento" (decisión de diseño: el salto seco se sentía brusco)
    spring.to(y, { response: Math.min(1.1, .55 + dist / 4000), damping:1, always:true }); // más lejos, un poco más largo
  };
  document.addEventListener('click', e => {
    // cualquier link a un lugar de esta misma página: "#kits" o "https://koopersound.com/#kits" estando en la home
    const a = e.target.closest('a[href*="#"]'); if(!a || a.target === '_blank') return;
    if(e.metaKey || e.ctrlKey || e.shiftKey || e.altKey) return;   // abrir en pestaña nueva sigue funcionando
    if(a.origin !== location.origin || a.pathname !== location.pathname || a.search !== location.search) return;
    const id = a.hash; if(id.length < 2) return;
    const el = document.getElementById(decodeURIComponent(id.slice(1))); if(!el) return;
    e.preventDefault();
    const head = document.querySelector('.header');
    const y = id === '#top' ? 0 : el.getBoundingClientRect().top + scrollY - (head ? head.offsetHeight : 0) - 12;
    history.replaceState(null, '', id);
    requestAnimationFrame(() => smoothScrollTo(y)); // después del toque (que corta cualquier scroll anterior)
  });
})();

/* ---------- carrusel de artistas: cambia solo y también se arrastra 1:1 ---------- */
(function(){
  const media = $('#heroMedia'), track = $('#heroTrack'), slides = $$('.slide'), dots = $$('.dot');
  // con una sola foto no hay carrusel: sin puntos, sin "Desliza" y sin arrastre
  if(slides.length < 2){ $('.hero-controls').style.display = 'none'; media.style.cursor = 'default'; return; }
  const n = slides.length; let index = 0, W = media.clientWidth, lastTouch = -Infinity;
  const render = x => {
    track.style.transform = `translate3d(${x}px,0,0)`;
    const p = -x / W; // los puntos acompañan el arrastre de forma continua
    dots.forEach((d, i) => {
      const k = Math.max(0, 1 - Math.abs(p - i));
      d.style.width = (8 + 20 * k) + 'px';
      d.firstChild.style.opacity = k;
    });
  };
  const spring = new Spring(0, render);
  const goTo = (i, opts = {}) => {
    const prev = index;
    index = ((i % n) + n) % n;
    slides.forEach((s, j) => s.setAttribute('aria-hidden', j !== index));
    dots.forEach((d, j) => d.setAttribute('aria-selected', j === index));
    if(reduceMotion.matches && prev !== index){ // sin deslizamiento: un fundido suave
      track.animate([{ opacity:0 }, { opacity:1 }], { duration:450, easing:'ease-out' });
    }
    spring.to(-index * W, { response:.5, damping:1, ...opts });
  };
  dots.forEach((d, i) => d.addEventListener('click', () => { lastTouch = performance.now(); goTo(i); }));
  addEventListener('resize', () => { W = media.clientWidth; spring.set(-index * W); });

  // gesto
  const vt = tracker(); let state = null, startX = 0, startY = 0, base = 0;
  media.addEventListener('pointerdown', e => {
    if(e.button !== 0) return;
    state = 'pending'; startX = e.clientX; startY = e.clientY; lastTouch = performance.now();
    vt.reset(); vt.add(e.clientX, e.timeStamp);
  });
  media.addEventListener('pointermove', e => {
    if(!state) return;
    const dx = e.clientX - startX, dy = e.clientY - startY;
    if(state === 'pending'){
      if(Math.abs(dy) > 10 && Math.abs(dy) > Math.abs(dx)){ state = null; return; } // es scroll vertical
      if(Math.abs(dx) < 10) return;                                              // umbral antes de decidir
      state = 'drag'; media.setPointerCapture(e.pointerId); media.classList.add('dragging');
      spring.stop(); base = spring.x; startX = e.clientX;                        // agarra desde donde está, sin salto
      $('#swipeHint').style.opacity = 0;
    }
    lastTouch = performance.now();
    vt.add(e.clientX, e.timeStamp);
    let x = base + (e.clientX - startX);
    const min = -(n - 1) * W;
    if(x > 0) x = rubberband(x, W);
    else if(x < min) x = min - rubberband(min - x, W);
    spring.x = x; spring.v = 0; render(x);
  });
  const end = () => {
    if(state !== 'drag'){ state = null; return; }
    state = null; media.classList.remove('dragging'); lastTouch = performance.now();
    const v = vt.v();
    const projected = spring.x + project(v);                       // a dónde va el gesto, no dónde se soltó
    const target = Math.max(0, Math.min(n - 1, Math.max(index - 1, Math.min(index + 1, Math.round(-projected / W)))));
    goTo(target, { velocity:v, damping: Math.abs(v) > 400 ? .82 : 1 }); // rebote solo si hubo impulso
  };
  media.addEventListener('pointerup', end);
  media.addEventListener('pointercancel', end);

  // cambio automático cada 5 s; si la persona la tocó, espera a que la suelte un rato
  setInterval(() => {
    if(document.hidden || state || performance.now() - lastTouch < 8000) return;
    goTo(index + 1);
  }, 5000);
  render(0); goTo(0);
})();

  btn.addEventListener('click', () => { audio.paused ? audio.play().catch(()=>{}) : audio.pause(); });

/* ---------- hoja del kit gratis: sube con resorte, se arrastra y se lanza para cerrar ---------- */

  }
  function close(velocity){
    isOpen = false; document.body.classList.remove('sheet-open');
    spring.to(H, { response:.38, damping:1, velocity, onRest: () => { if(!isOpen) show(false); } });
    lastFocus && lastFocus.focus && lastFocus.focus({ preventScroll:true });
  }
  $$('[data-open-sheet]').forEach(b => b.addEventListener('click', open));

      dragging = true; moved = false; startY = e.clientY; spring.stop(); base = spring.x;

    el.addEventListener('pointerup', end);
